# Void Reading Surface Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Restore reliable offline ArchWiki article rendering and refine the primary reading surfaces into a quieter, more legible Violet Void instrument panel.

**Architecture:** Protect the visual behavior with source-to-built-CSS contract tests, then make focused Stylus changes at the responsible component boundaries. Preserve shared radius/shadow tokens and semantic terminal colours; validate the canonical UserCSS in a real offline Vector document at desktop and mobile viewports.

**Tech Stack:** Stylus, Node.js 20+, `node:test`, PostCSS, Playwright Chromium, Arch `arch-wiki-docs`

## Global Constraints

- Palette: Void `#0f0f0f`, Base `#181818`, Raised `#202020`, Text `#e7e7e7`, Muted Link `#b3a7d8`, Signal Violet `#8950c7`.
- Body and headings use the existing Inter/system stack; code and utility text use JetBrains Mono/Fira Code fallbacks.
- Reading progress must never transform `.mw-parser-output`.
- Generic checkbox styling must exclude `.vector-dropdown-checkbox` while ordinary checkbox and radio styling remains intact.
- Desktop keeps a stable article plus restrained left TOC; `html.client-nojs` mobile uses a compact, scrollable title-bar TOC.
- Primary `.wikitable`, `.ambox`, `.ombox`, `.imbox`, and `.tmbox` surfaces become flat; modal, dropdown, tooltip, category/footer, and other elevated surfaces remain unchanged.
- Do not change shared layout radius or shadow tokens in `src/variables/layout.styl`.
- Preserve the existing untracked `output/` screenshots until the explicit recapture step.

- [ ] **Step A1: Add rule helpers below `readProducedCss()`**

```js
function rulesContaining(root, selectorFragment) {
  return root.nodes
    .flatMap((node) => (node.type === 'atrule' ? node.nodes || [] : [node]))
    .filter(
      (node) => node.type === 'rule' && node.selector.includes(selectorFragment)
    )
}

function declarationValue(rule, property) {
  return rule.nodes.find(
    (node) => node.type === 'decl' && node.prop === property
  )?.value
}
```

- [ ] **Step A2: Add the failing canonical-CSS test**

```js
test('void reading surface contracts are emitted in canonical CSS', () => {
  const rootDir = createFixture()
  try {
    runBuild(rootDir)
    const { css } = readProducedCss(rootDir)
    const documentRoot = postcss.parse(css)
    const scope = documentRoot.nodes.find(
      (node) => node.type === 'atrule' && node.name === '-moz-document'
    )
    const root = { nodes: scope.nodes }

    assert.equal(
      rulesContaining(root, '.mw-parser-output').some(
        (rule) => declarationValue(rule, 'animation-name') === 'reading-progress'
      ),
      false,
      'article content must not be the reading-progress target'
    )
    assert.match(
      css,
      /input\[type=["']checkbox["']\]:not\(\.vector-dropdown-checkbox\)/
    )
    assert.match(
      css,
      /\.vector-toc \.vector-toc-text\s*\{[^}]*word-break:\s*normal\s*!important;[^}]*overflow-wrap:\s*normal\s*!important;/s
    )
    assert.match(
      css,
      /\.vector-toc-list-item:has\(> \.vector-toc-toggle\)\s*>\s*\.vector-toc-link/
    )
    assert.match(
      css,
      /html\.client-nojs \.vector-page-titlebar-toc \.vector-dropdown-content\s*\{[^}]*width:\s*min\(22rem,calc\(100vw - 2rem\)\)\s*!important;[^}]*max-height:\s*min\(60vh,28rem\);[^}]*overflow-y:\s*auto;/s
    )
    assert.match(css, /html body \.mw-parser-output a\s*\{[^}]*color:\s*#b3a7d8\s*!important;/s)
    assert.match(css, /html body \.mw-parser-output \.archwiki-template-(?:man|pkg) a[^}]*color:\s*#42ff97\s*!important/s)

    assert.ok(
      rulesContaining(root, '.wikitable').some(
        (rule) => declarationValue(rule, 'box-shadow') === 'none' &&
          declarationValue(rule, 'border-radius') === '0'
      ),
      'primary wikitable must be flat'
    )
    for (const selector of ['.ambox', '.ombox', '.imbox', '.tmbox']) {
      assert.ok(
        rulesContaining(root, selector).some(
          (rule) => declarationValue(rule, 'box-shadow') === 'none'
        ),
        `${selector} must be flat`
      )
    }
    assert.ok(
      rulesContaining(root, 'h1::after').some(
        (rule) => declarationValue(rule, 'background') === '#8950c7'
      )
    )
    assert.ok(
      rulesContaining(root, 'h2').some(
        (rule) =>
          declarationValue(rule, 'border-bottom') ===
          '1px solid rgba(137,80,199,0.35)'
      )
    )
  } finally {
    removeFixture(rootDir)
  }
})
```

- [ ] **Step D2: Run this Playwright snapshot-first probe from the repository root**

```bash
node <<'NODE'
const assert = require('node:assert/strict')
const fs = require('node:fs')
const path = require('node:path')
const { chromium } = require('playwright')
const { extractScopedCss } = require('./scripts/lib/userstyle-test-css')

const css = extractScopedCss(
  fs.readFileSync('dist/violet-void-theme-archwiki.user.css', 'utf8')
)
const outputDir = 'output/playwright/offline-archwiki'
fs.mkdirSync(outputDir, { recursive: true })
const pages = [
  ['installation-guide', 'Installation_guide.html'],
  ['wireguard', 'WireGuard.html'],
]
const viewports = [
  ['desktop', { width: 1440, height: 1000 }],
  ['mobile', { width: 390, height: 844 }],
]

;(async () => {
  const browser = await chromium.launch({ executablePath: '/usr/sbin/chromium' })
  try {
    for (const [pageName, file] of pages) {
      for (const [mode, viewport] of viewports) {
        const context = await browser.newContext({ viewport })
        const page = await context.newPage()
        await page.goto(`http://127.0.0.1:4173/en/${file}`, {
          waitUntil: 'domcontentloaded',
        })
        await page.addStyleTag({ content: css })
        await page.screenshot({
          path: path.join(outputDir, `${pageName}-${mode}.png`),
          fullPage: false,
        })
        const result = await page.evaluate(() => {
          const parser = document.querySelector('.mw-parser-output')
          const paragraph = parser?.querySelector('p')
          const firstLink = [...(parser?.querySelectorAll('a:not(.new)') || [])]
            .find((link) =>
              !link.closest('.archwiki-template-man, .archwiki-template-pkg')
            )
          const tocText = document.querySelector('.vector-toc-text')
          const h1 = document.querySelector('h1')
          const table = document.querySelector('.wikitable')
          const message = document.querySelector('.ambox, .ombox, .imbox, .tmbox')
          const badSquares = [...document.querySelectorAll('.vector-dropdown-checkbox')]
            .filter((element) => {
              const style = getComputedStyle(element)
              return style.display !== 'none' && style.visibility !== 'hidden' &&
                style.opacity !== '0' && style.borderTopWidth !== '0px'
            }).length
          return {
            parserWidth: parser?.getBoundingClientRect().width || 0,
            paragraphWidth: paragraph?.getBoundingClientRect().width || 0,
            articleBackground: parser && getComputedStyle(parser).backgroundColor,
            articleText: paragraph && getComputedStyle(paragraph).color,
            linkColor: firstLink && getComputedStyle(firstLink).color,
            h1Rule: h1 && getComputedStyle(h1, '::after').backgroundColor,
            tocWordBreak: tocText && getComputedStyle(tocText).wordBreak,
            tocOverflowWrap: tocText && getComputedStyle(tocText).overflowWrap,
            badSquares,
            horizontalOverflow: document.documentElement.scrollWidth >
              document.documentElement.clientWidth,
            tableShadow: table ? getComputedStyle(table).boxShadow : 'none',
            tableRadius: table ? getComputedStyle(table).borderRadius : '0px',
            messageImage: message ? getComputedStyle(message).backgroundImage : 'none',
            messageShadow: message ? getComputedStyle(message).boxShadow : 'none',
          }
        })
        assert.ok(result.parserWidth > 0)
        assert.ok(result.paragraphWidth > 0)
        assert.equal(result.articleBackground, 'rgb(24, 24, 24)')
        assert.equal(result.articleText, 'rgb(231, 231, 231)')
        assert.equal(result.linkColor, 'rgb(179, 167, 216)')
        assert.equal(result.h1Rule, 'rgb(137, 80, 199)')
        assert.equal(result.tocWordBreak, 'normal')
        assert.equal(result.tocOverflowWrap, 'normal')
        assert.equal(result.badSquares, 0)
        assert.equal(result.horizontalOverflow, false)
        assert.equal(result.tableShadow, 'none')
        assert.equal(result.tableRadius, '0px')
        assert.equal(result.messageImage, 'none')
        assert.equal(result.messageShadow, 'none')
        console.log(pageName, mode, result)
        await context.close()
      }
    }
  } finally {
    await browser.close()
  }
})().catch((error) => {
  console.error(error)
  process.exitCode = 1
})
NODE
```
